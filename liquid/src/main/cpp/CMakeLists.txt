cmake_minimum_required(VERSION 3.22)
project(liquid_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(LIQUID_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/llama.cpp-liquid" ABSOLUTE)
if(NOT EXISTS "${LIQUID_ROOT}/tools/liquid-audio/runner.h")
 message(FATAL_ERROR "Run python3 scripts/prepare_liquid.py first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(Vulkan_INCLUDE_DIR "${LIQUID_ROOT}/../Vulkan-Headers/include" CACHE PATH "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIQUID_ROOT}" upstream EXCLUDE_FROM_ALL)
# App-local Vulkan integration: serialize the vendor compiler and record pipeline
# names before entering driver code (a SIGSEGV cannot be caught as vk::SystemError).
file(READ "${LIQUID_ROOT}/ggml/src/ggml-vulkan/ggml-vulkan.cpp" VULKAN_SOURCE)
set(VK_START_ANCHOR "    vk::ShaderModuleCreateInfo shader_module_create_info({}, spv_size, reinterpret_cast<const uint32_t *>(spv_data));")
set(VK_END_ANCHOR "    pipeline->compiled = true;")
foreach(ANCHOR IN ITEMS "${VK_START_ANCHOR}" "${VK_END_ANCHOR}")
 string(FIND "${VULKAN_SOURCE}" "${ANCHOR}" ANCHOR_INDEX)
 if(ANCHOR_INDEX EQUAL -1)
  message(FATAL_ERROR "Pinned Vulkan integration anchor missing")
 endif()
endforeach()
string(REPLACE "${VK_START_ANCHOR}" [=[
    static std::mutex android_pipeline_compile_mutex;
    std::lock_guard<std::mutex> android_pipeline_compile_lock(android_pipeline_compile_mutex);
    fprintf(stderr, "vulkan_pipeline_begin name=%s bytes=%zu subgroup=%u\n", pipeline->name.c_str(), spv_size, required_subgroup_size);
    fflush(stderr);
    vk::ShaderModuleCreateInfo shader_module_create_info({}, spv_size, reinterpret_cast<const uint32_t *>(spv_data));
]=] VULKAN_SOURCE "${VULKAN_SOURCE}")
string(REPLACE "${VK_END_ANCHOR}" [=[
    pipeline->compiled = true;
    fprintf(stderr, "vulkan_pipeline_end name=%s\n", pipeline->name.c_str());
    fflush(stderr);
]=] VULKAN_SOURCE "${VULKAN_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ggml-vulkan-android.cpp" "// App-local compiler compatibility and diagnostics.\n${VULKAN_SOURCE}")
get_target_property(VK_SOURCES ggml-vulkan SOURCES)
list(FIND VK_SOURCES "ggml-vulkan.cpp" VK_SOURCE_INDEX)
if(VK_SOURCE_INDEX EQUAL -1)
 message(FATAL_ERROR "Pinned Vulkan source list changed")
endif()
list(REMOVE_ITEM VK_SOURCES ggml-vulkan.cpp)
list(APPEND VK_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/ggml-vulkan-android.cpp")
set_property(TARGET ggml-vulkan PROPERTY SOURCES "${VK_SOURCES}")
target_include_directories(ggml-vulkan PRIVATE "${LIQUID_ROOT}/ggml/src/ggml-vulkan")
set(LLAMA_INSTALL_VERSION "0.0.0")
add_subdirectory("${LIQUID_ROOT}/tools/mtmd" mtmd EXCLUDE_FROM_ALL)
# App-local integration copy: explicitly control the detokenizer's device and threads.
# Preserve the pinned dependency checkout unchanged.
file(READ "${LIQUID_ROOT}/tools/mtmd/audio-decoder.cpp" DECODER_SOURCE)
string(FIND "${DECODER_SOURCE}" "common_params params_audio_tokenizer;" DECODER_ANCHOR)
if(DECODER_ANCHOR EQUAL -1)
 message(FATAL_ERROR "Audio detokenizer integration anchor missing")
endif()
string(REPLACE "common_params params_audio_tokenizer;" "common_params params_audio_tokenizer;\n        params_audio_tokenizer.n_gpu_layers = use_gpu ? 99 : 0;\n        params_audio_tokenizer.no_op_offload = !use_gpu;\n        params_audio_tokenizer.no_kv_offload = !use_gpu;\n        if (!use_gpu) params_audio_tokenizer.devices = {nullptr};\n        params_audio_tokenizer.cpuparams.n_threads = n_threads;\n        params_audio_tokenizer.cpuparams_batch.n_threads = n_threads;" DECODER_SOURCE "${DECODER_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/audio-decoder-android.cpp" "// Modified for Android benchmark device/thread selection.\n${DECODER_SOURCE}")
get_target_property(MTMD_SOURCES mtmd SOURCES)
list(REMOVE_ITEM MTMD_SOURCES audio-decoder.cpp)
list(APPEND MTMD_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/audio-decoder-android.cpp")
set_property(TARGET mtmd PROPERTY SOURCES "${MTMD_SOURCES}")
add_library(liquid_runner STATIC "${LIQUID_ROOT}/tools/liquid-audio/runner.cpp")
target_include_directories(liquid_runner PUBLIC "${LIQUID_ROOT}/tools/liquid-audio")
target_link_libraries(liquid_runner PUBLIC llama common mtmd)
add_library(liquid_bench SHARED benchmark.cpp)
target_include_directories(liquid_bench PRIVATE "${LIQUID_ROOT}/vendor")
target_link_libraries(liquid_bench PRIVATE liquid_runner android log)
target_link_options(liquid_bench PRIVATE "-Wl,-z,max-page-size=16384")
