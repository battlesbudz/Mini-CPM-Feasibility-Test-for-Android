cmake_minimum_required(VERSION 3.22)
project(moshi_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MOSHI_HOST_SMOKE "Build host codec tests" OFF)
option(MOSHI_HOST_VULKAN "Enable Vulkan in host tests" OFF)
get_filename_component(VENDOR "${CMAKE_CURRENT_LIST_DIR}/../../../../vendor" ABSOLUTE)
if(NOT EXISTS "${VENDOR}/moshi-android-src/src/moshi.cpp")
    message(FATAL_ERROR "Run python3 scripts/prepare_moshi.py first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(MOSHI_HOST_SMOKE AND NOT MOSHI_HOST_VULKAN)
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
else()
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
endif()
set(Vulkan_INCLUDE_DIR "${VENDOR}/Vulkan-Headers-moshi/include" CACHE PATH "" FORCE)
add_subdirectory("${VENDOR}/ggml-moshi-android" ggml EXCLUDE_FROM_ALL)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
add_subdirectory("${VENDOR}/sentencepiece-moshi" sentencepiece EXCLUDE_FROM_ALL)
add_library(moshi_core STATIC
    tensor_trace.cpp
    codec_precision.cpp
    "${VENDOR}/moshi-android-src/src/moshi.cpp"
    "${VENDOR}/moshi-android-src/src/json.cpp"
    "${VENDOR}/moshi-android-src/src/safetensor.cpp")
target_include_directories(moshi_core PRIVATE "${VENDOR}/ggml-moshi-android/src")
target_include_directories(moshi_core PUBLIC
    "${CMAKE_CURRENT_LIST_DIR}"
    "${VENDOR}/moshi-android-src/include"
    "${VENDOR}/sentencepiece-moshi/src")
target_link_libraries(moshi_core PUBLIC ggml sentencepiece-static)
if(MOSHI_HOST_SMOKE)
    add_executable(moshi_im2col_test ../../test/cpp/im2col_test.cpp)
    target_include_directories(moshi_im2col_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
    target_link_libraries(moshi_im2col_test PRIVATE ggml)
    if(MOSHI_HOST_VULKAN)
        target_compile_definitions(moshi_im2col_test PRIVATE MOSHI_TEST_VULKAN=1)
    endif()
    add_executable(moshi_trace_test ../../test/cpp/tensor_trace_test.cpp)
    target_include_directories(moshi_trace_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${VENDOR}/ggml-moshi-android/src")
    target_link_libraries(moshi_trace_test PRIVATE ggml)
    add_executable(moshi_precision_test ../../test/cpp/precision_test.cpp)
    target_link_libraries(moshi_precision_test PRIVATE moshi_core)
    if(MOSHI_HOST_VULKAN)
        target_compile_definitions(moshi_precision_test PRIVATE MOSHI_TEST_VULKAN=1)
        # Fault injection exists only in host tests, never in the Android APK.
        target_compile_definitions(ggml-vulkan PRIVATE MOSHI_VULKAN_TEST_FAILURE=1)
        add_executable(moshi_q4_test ../../test/cpp/q4_test.cpp)
        target_include_directories(moshi_q4_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
        target_link_libraries(moshi_q4_test PRIVATE ggml)
    endif()
    add_executable(moshi_codec_smoke ../../test/cpp/codec_smoke.cpp)
    target_link_libraries(moshi_codec_smoke PRIVATE moshi_core)
else()
    add_library(moshi_bench SHARED benchmark.cpp)
    target_link_libraries(moshi_bench PRIVATE moshi_core android log)
    target_link_options(moshi_bench PRIVATE "-Wl,-z,max-page-size=16384")
endif()
